### CONFIGURATION PILLAR 3: SERVER-SIDE MONOTONIC TELEMETRY GUARDRAILS
Create `lib/validators/ecuPayloadGuard.ts`:
```typescript
export interface ValidationMetrics {
  isMonotonic: boolean;
  detectedRows: number;
  detectedCols: number;
  unitSystemVerified: boolean;
  errors: string[];
}

export interface IngestedPayload {
  xAxis: number[];
  yAxis: number[];
  dataMatrix: number[][];
  metrics: ValidationMetrics;
}

export class EcuPayloadGuard {
  private static verifyArrayMonotonicity(array: number[]): boolean {
    if (array.length < 2) return true;
    const initialDelta = array[1] - array[0];
    if (initialDelta === 0) return false;
    
    for (let i = 1; i < array.length - 1; i++) {
      const currentDelta = array[i + 1] - array[i];
      if (initialDelta > 0 && currentDelta <= 0) return false;
      if (initialDelta < 0 && currentDelta >= 0) return false;
    }
    return true;
  }

  public static processTelemetryStream(rawCsvData: string, expectedXUnit: string, expectedYUnit: string): IngestedPayload {
    const errorLogs: string[] = [];
    const lines = rawCsvData.split(/\r?\n/).map(line => line.trim()).filter(line => line.length > 0);
    
    if (lines.length < 3) {
      throw new Error("Telemetry payload structurally invalid: Minimum grid limits not met.");
    }

    const headerMetadata = lines[0].toLowerCase();
    if (!headerMetadata.includes(expectedXUnit.toLowerCase()) || !headerMetadata.includes(expectedYUnit.toLowerCase())) {
      errorLogs.push(`Unit validation variant warning: Map requires explicit matching contexts for ${expectedXUnit} and ${expectedYUnit}.`);
    }

    const xAxisCoordinates = lines[1].split(',').slice(1).map(val => {
      const parsed = parseFloat(val);
      if (isNaN(parsed)) throw new Error("Matrix parsing fault: Non-numeric token found on horizontal axis reference point.");
      return parsed;
    });

    const yAxisCoordinates: number[] = [];
    const matrixValues: number[][] = [];

    for (let i = 2; i < lines.length; i++) {
      const segmentRow = lines[i].split(',');
      const rowYCoordinate = parseFloat(segmentRow[0]);
      if (isNaN(rowYCoordinate)) throw new Error(`Matrix parsing fault: Non-numeric token found on vertical coordinate mapping index ${i}.`);
      
      yAxisCoordinates.push(rowYCoordinate);
      
      const rowMatrixValues = segmentRow.slice(1).map((val, colIdx) => {
        const parsedValue = parseFloat(val);
        if (isNaN(parsedValue)) {
          throw new Error(`Data corruption trace located at intersection row ${i}, column ${colIdx}.`);
        }
        return parsedValue;
      });

      if (rowMatrixValues.length !== xAxisCoordinates.length) {
        throw new Error(`Asymmetrical structure detected at entry sequence ${i}: Width array bounds do not compute.`);
      }
      matrixValues.push(rowMatrixValues);
    }

    const xIsMonotonic = this.verifyArrayMonotonicity(xAxisCoordinates);
    const yIsMonotonic = this.verifyArrayMonotonicity(yAxisCoordinates);

    if (!xIsMonotonic || !yIsMonotonic) {
      errorLogs.push("Mathematical matrix deadlock: Non-monotonic coordinate parameters discovered inside telemetry stream.");
    }

    return {
      xAxis: xAxisCoordinates,
      yAxis: yAxisCoordinates,
      dataMatrix: matrixValues,
      metrics: {
        isMonotonic: xIsMonotonic && yIsMonotonic,
        detectedRows: yAxisCoordinates.length,
        detectedCols: xAxisCoordinates.length,
        unitSystemVerified: errorLogs.length === 0,
        errors: errorLogs
      }
    };
  }
}
```
